cmake_minimum_required(VERSION 3.22)
project(DockFolderBackdrop LANGUAGES CXX)
include(identity.cmake)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_SKIP_RPATH ON)
set(KWIN_BACKEND "wayland" CACHE STRING "KWin backend: wayland or x11")
if(KWIN_BACKEND STREQUAL "x11")
    set(KWIN_NAMESPACE kwin-x11)
else()
    set(KWIN_NAMESPACE kwin)
endif()
find_package(Qt6 REQUIRED COMPONENTS Gui Quick DBus)
find_path(KCOREADDONS_INCLUDE_DIR KPluginFactory PATH_SUFFIXES KF6/KCoreAddons REQUIRED)
find_path(KCONFIG_INCLUDE_DIR KSharedConfig PATH_SUFFIXES KF6/KConfigCore REQUIRED)
find_path(KCONFIG_VERSION_INCLUDE_DIR kconfig_version.h PATH_SUFFIXES KF6/KConfig REQUIRED)
find_library(KCOREADDONS_LIBRARY NAMES KF6CoreAddons REQUIRED)
find_library(KCONFIG_LIBRARY NAMES KF6ConfigCore REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(EPOXY REQUIRED IMPORTED_TARGET epoxy)
find_path(KWIN_INCLUDE_DIR effect/effect.h PATH_SUFFIXES ${KWIN_NAMESPACE} REQUIRED)
find_library(KWIN_LIBRARY NAMES ${KWIN_NAMESPACE} REQUIRED)
add_library(dockfoldersbackdrop MODULE backdrop.cpp gaussian.cpp contrast.cpp)
set_target_properties(dockfoldersbackdrop PROPERTIES PREFIX "" OUTPUT_NAME "${DOCK_BACKDROP_PLUGIN}")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/plugin-name.txt" CONTENT "${DOCK_BACKDROP_PLUGIN}\n")
target_include_directories(dockfoldersbackdrop PRIVATE ${KWIN_INCLUDE_DIR} ${KCOREADDONS_INCLUDE_DIR}
    ${KCONFIG_INCLUDE_DIR} ${KCONFIG_VERSION_INCLUDE_DIR})
target_link_libraries(dockfoldersbackdrop PRIVATE ${KWIN_LIBRARY} Qt6::Gui Qt6::Quick Qt6::DBus
    ${KCOREADDONS_LIBRARY} ${KCONFIG_LIBRARY} PkgConfig::EPOXY)
target_compile_options(dockfoldersbackdrop PRIVATE -Wall -Wextra -Werror)
install(TARGETS dockfoldersbackdrop LIBRARY DESTINATION ${KWIN_NAMESPACE}/effects/plugins)
option(DOCK_BLUR_TEST_FIXTURES "Build private compositor test effects" OFF)
if(DOCK_BLUR_TEST_FIXTURES)
    add_library(inheritedglstate MODULE ../../tests/native/inherited-gl-state.cpp)
    set_target_properties(inheritedglstate PROPERTIES PREFIX "")
    target_include_directories(inheritedglstate PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}
        ${KWIN_INCLUDE_DIR} ${KCOREADDONS_INCLUDE_DIR} ${KCONFIG_INCLUDE_DIR} ${KCONFIG_VERSION_INCLUDE_DIR})
    target_link_libraries(inheritedglstate PRIVATE ${KWIN_LIBRARY} Qt6::Gui Qt6::Quick
        ${KCOREADDONS_LIBRARY} ${KCONFIG_LIBRARY} PkgConfig::EPOXY)
    target_compile_options(inheritedglstate PRIVATE -Wall -Wextra -Werror)
    install(TARGETS inheritedglstate LIBRARY DESTINATION ${KWIN_NAMESPACE}/effects/plugins)
    add_library(presentedframe MODULE ../../tests/native/presented-frame.cpp)
    set_target_properties(presentedframe PROPERTIES PREFIX "")
    target_include_directories(presentedframe PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}
        ${KWIN_INCLUDE_DIR} ${KCOREADDONS_INCLUDE_DIR} ${KCONFIG_INCLUDE_DIR} ${KCONFIG_VERSION_INCLUDE_DIR})
    target_link_libraries(presentedframe PRIVATE ${KWIN_LIBRARY} Qt6::Gui Qt6::Quick
        ${KCOREADDONS_LIBRARY} ${KCONFIG_LIBRARY} PkgConfig::EPOXY)
    target_compile_options(presentedframe PRIVATE -Wall -Wextra -Werror)
    install(TARGETS presentedframe LIBRARY DESTINATION ${KWIN_NAMESPACE}/effects/plugins)
endif()
